function store_config() {
    return [
        'rylami' => ['name' => 'Rylami', 'base' => 'https://rylami.com', 'resources' => '/resources.html'],
        'beacon' => ['name' => 'Beacon of Hope Counseling', 'base' => 'https://www.beaconofhopenc.com', 'resources' => '/resources.html'],

function store_info($slug) {
    $stores = store_config();
    if (!isset($stores[$slug])) { throw new RuntimeException('Unknown storefront.'); }
    return $stores[$slug];
}

function store_db() {
    $path = '/home/nhnqsp8jasbe/bohc-db-config.php';
    if (!is_readable($path)) { throw new RuntimeException('Database configuration unavailable.'); }
    $db = require $path;
    return new PDO('mysql:host=' . ($db['host'] ?? 'localhost') . ';dbname=' . ($db['database'] ?? '') . ';charset=utf8mb4', $db['username'] ?? '', $db['password'] ?? '', [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC, PDO::ATTR_EMULATE_PREPARES => false]);
}
function e($value) { return htmlspecialchars((string)$value, ENT_QUOTES, 'UTF-8'); }
function store_headers() {
    header('Cache-Control: private, no-store, max-age=0');
    header('Referrer-Policy: no-referrer');
    header('X-Content-Type-Options: nosniff');
    header('X-Robots-Tag: noindex, nofollow, noarchive');
}
function store_start_page($store, $title, $refresh = false) {
    store_headers();
    echo '<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta name="robots" content="noindex,nofollow,noarchive">';
    if ($refresh) { echo '<meta http-equiv="refresh" content="5">'; }
    echo '<title>' . e($title . ' | ' . $store['name']) . '</title><style>body{margin:0;background:#fffdf8;color:#354a3d;font:18px/1.6 system-ui,sans-serif}header,main{max-width:760px;margin:auto;padding:24px}main{background:white;border:1px solid #dcded4;border-radius:16px;margin:24px auto}a{color:#386c7b}.button{display:inline-block;padding:14px 22px;background:#354a3d;color:white;border-radius:6px;text-decoration:none;margin:10px 0}h1{font-family:Georgia,serif;line-height:1.2}.note{font-size:15px}li{margin:12px 0}@media(max-width:800px){main{margin:16px}}</style></head><body><header><a href="' . e($store['base']) . '">' . e($store['name']) . '</a></header><main>';
}
